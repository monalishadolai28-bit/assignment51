#include <stdio.h>

int main() {
    int n = 3; // Size of the matrix
    int a[n][n];
    int i, j;
    int num = 1;
    int top = 0, bottom = n-1, left = 0, right = n-1;

    while(num <= n*n) {
        // Top row
        for(i = left; i <= right; i++)
            a[top][i] = num++;
        top++;

        // Right column
        for(i = top; i <= bottom; i++)
            a[i][right] = num++;
        right--;

        // Bottom row
        for(i = right; i >= left; i--)
            a[bottom][i] = num++;
        bottom--;

        // Left column
        for(i = bottom; i >= top; i--)
            a[i][left] = num++;
        left++;
    }

    // Print the matrix
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            printf("%d ", a[i][j]);
        }
        printf("\n");
    }

    return 0;
}
